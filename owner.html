Arduino C++ Controller Pin configuration 
#include <ESP8266WiFi.h>
#include <Servo.h>
#include <SPI.h>
#include <MFRC522.h>

#define TRIG_PIN_1 D0
#define ECHO_PIN_1 D1
#define SERVO_PIN  D4
#define SS_PIN     D8
#define RST_PIN    D3

MFRC522 rfid(SS_PIN, RST_PIN);
Servo gateServo;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN_1, OUTPUT);
  pinMode(ECHO_PIN_1, INPUT);
  SPI.begin();
  rfid.PCD_Init();
  gateServo.attach(SERVO_PIN);
  gateServo.write(0);
}
